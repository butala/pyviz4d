built extension, which ``make test`` builds. Nothing here imports vtk or

def test_sphericalct_theta_is_latitude_not_colatitude():
    """The functional test: this is what a convention clash breaks.

    ``rtp2ijk`` is the C++'s own point-to-voxel map. It accepts any
    ``theta in [-pi/2, pi/2]`` and rejects anything outside that, which is a
    latitude range; a colatitude grid would run 0 .. pi and accept 3.0 rad. The
    equator lands in the middle bin and the south pole in bin 0, both of which
    only follow from ``z = r sin(theta)``.
    """
    hs = UniformHollowSphere(N_RAD, N_THETA, N_PHI, R_MIN, R_MAX)
    assert tuple(hs.xyz2ijk(DoubleVector3(2.5, 0.0, 0.0)))[1] == N_THETA // 2
    assert tuple(hs.xyz2ijk(DoubleVector3(0.0, 0.0, -2.5)))[1] == 0
    # +20 deg latitude is in the bin the arithmetic predicts. (The north pole is
    # NOT -- see the known-defect test below.)
    p = OD.sph2cart(2.7, 20.0, 0.0)
    assert tuple(hs.xyz2ijk(DoubleVector3(*[float(v) for v in p])))[1] == \
        int(np.floor((np.radians(20.0) + np.pi / 2) / (np.pi / N_THETA)))
    # A legal colatitude, and an illegal latitude: rejected.
    with pytest.raises((RuntimeError, ValueError)):
        hs.rtp2ijk(DoubleVector3(2.7, 3.0, 0.0))
